
import tkinter as tk

# Create the main window
root = tk.Tk()
root.title("My Calculator")
root.geometry("320x450")
root.resizable(False, False)

# Display screen
display = tk.Entry(
    root,
    font=("Arial", 24),
    justify="right",
    bd=10,
    relief=tk.GROOVE
)
display.pack(fill="x", padx=10, pady=10)


# Function to handle button clicks
def click(value):
    display.insert(tk.END, value)


# Function to clear the screen
def clear():
    display.delete(0, tk.END)


# Function to calculate the answer
def calculate():
    try:
        expression = display.get()

        # Allow only calculator characters
        allowed = "0123456789+-*/.() "

        if not expression or any(c not in allowed for c in expression):
            display.delete(0, tk.END)
            display.insert(0, "Invalid input")
            return

        result = eval(expression, {"__builtins__": {}}, {})

        display.delete(0, tk.END)
        display.insert(0, str(result))

    except ZeroDivisionError:
        display.delete(0, tk.END)
        display.insert(0, "Cannot divide by zero")

    except Exception:
        display.delete(0, tk.END)
        display.insert(0, "Error")


# Calculator buttons
buttons = [
    ["7", "8", "9", "/"],
    ["4", "5", "6", "*"],
    ["1", "2", "3", "-"],
    ["C", "0", ".", "+"],
    ["(", ")", "=", "⌫"]
]

# Create button layout
for row in buttons:
    frame = tk.Frame(root)
    frame.pack(expand=True, fill="both")

    for item in row:
        if item == "C":
            command = clear
        elif item == "=":
            command = calculate
        elif item == "⌫":
            command = lambda: display.delete(
                len(display.get()) - 1, tk.END
            ) if display.get() else None
        else:
            command = lambda value=item: click(value)

        button = tk.Button(
            frame,
            text=item,
            font=("Arial", 16),
            command=command
        )
        button.pack(
            side="left",
            expand=True,
            fill="both",
            padx=2,
            pady=2
        )

# Run the calculator
root.mainloop()
